const type Address = {
  postalCode: string; // 우편번호
  province: string; // 시/도 (예: 서울특별시, 경기도)
  city: string; // 구/군/구 (예: 강남구, 수원시)
  district: string; // 동/읍/면 (예: 역삼동, 원천동)
  streetName: string; // 도로명 (예: 테헤란로)
  buildingNumber: string; // 건물번호 (예: 123, 101호)
  detailAddress?: string; // 상세주소 (선택사항, 예: 아파트 101동 201호)
  country: string; // 국가 (예: 대한민국)
}

const type Customer = {
 customerId : string
 password : string
 name : string
 email : string
 tel : string
 address : Address
 prefer : string //선호 식품 *입력 가능한 값을 Admin.category와 동일한 값들로 제한할것
 joinDate : Date
 inactive : string //휴면여부
}

const type Admin = {

 operatingCloseTime : string //영업종료시간
 joinDate : Date //가입 일자
 inactive : string //휴면 여부
}
